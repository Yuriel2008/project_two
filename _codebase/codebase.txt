session_start();
// Basic Auth Check (In a real app, check for 'is_admin' role)
if (!isset($_SESSION['user_id']) || $_SESSION['role'] !== 'admin') {
  // For prototype, we assume if logged in, you can access.

// --- 2. DATABASE CONNECTION ---
$dbHost = 'localhost';
$dbName = 'lingoflow_db';
$dbUser = 'root';
$dbPass = ''; // Update with your password
try {
  $pdo = new PDO("mysql:host=$dbHost;dbname=$dbName;charset=utf8mb4", $dbUser, $dbPass);
  $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
  // --- 3. HANDLE ACTIONS (POST) ---
  // DELETE LESSON
  if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'delete') {
    $idToDelete = $_POST['id'];
    $deleteStmt = $pdo->prepare("DELETE FROM lessons WHERE id = ?");
    $deleteStmt->execute([$idToDelete]);
    header("Location: content.php");// Refresh to clear post data
    exit;
  }
  // ADD LESSON
  if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action']) && $_POST['action'] === 'add') {
    $newId = $_POST['lesson_id'];
    $newTitle = $_POST['title'];
    $newCat = $_POST['category'];
    $newXp = $_POST['xp'];
    $newOrder = $_POST['sort_order'];
    $addStmt = $pdo->prepare("INSERT INTO lessons (id, title, category, description, xp_value, sort_order) VALUES (?, ?, ?, 'New content', ?, ?)");
    $addStmt->execute([
      $newId,
      $newTitle,
      $newCat,
      $newXp,
      $newOrder
    ]);
    header("Location: content.php");// Refresh to clear post data
    exit;
  }
  // --- 4. FETCH DATA ---
  // Stats
  $statsStmt = $pdo->query("SELECT COUNT(*) as count, SUM(xp_value) as total_xp FROM lessons");
  $stats = $statsStmt->fetch(PDO::FETCH_ASSOC);
  $totalLessons = $stats['count'];
  $totalXP = $stats['total_xp'];
  // Search Filter
  $search = $_GET['q'] ?? '';
  $sql = "SELECT * FROM lessons";
  $params = [];
  if ($search) {
    $sql .= " WHERE title LIKE ? OR category LIKE ?";
    $params = ["%$search%", "%$search%"];
  }
  $sql .= " ORDER BY sort_order ASC";
  $stmt = $pdo->prepare($sql);
  $stmt->execute($params);
  $lessons = $stmt->fetchAll(PDO::FETCH_ASSOC);
} catch (PDOException $e) {
  die("Database connection failed: " . $e->getMessage());
}

<?php
// --- 1. SESSION & AUTHENTICATION ---
session_start();
// Redirect to login if session is missing
if (!isset($_SESSION['user_id'])) {
  header("Location: login.php");
  exit;
}
$user_id = $_SESSION['user_id'];
// --- 2. DATABASE CONNECTION (Self-Contained) ---
$dbHost = 'localhost';
$dbName = 'lingoflow_db';
$dbUser = 'root';
$dbPass = ''; // Update with your password if needed
try {
  $pdo = new PDO("mysql:host=$dbHost;dbname=$dbName;charset=utf8mb4", $dbUser, $dbPass);
  $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
  // --- 3. FETCH USER STATS ---
  $stmt = $pdo->prepare("SELECT username, xp, streak, accuracy, rank FROM users WHERE id = ?");
  $stmt->execute([$user_id]);
  $user = $stmt->fetch(PDO::FETCH_ASSOC);
  if (!$user) {
    // User invalid/deleted, kill session
    session_destroy();
    header("Location: login.php");
    exit;
  }
  // Calculate Level (Simple logic: 1 Level per 200 XP)
  $level = floor($user['xp'] / 200) + 1;
  // --- 4. FETCH NEXT ACTIVE LESSON ---
  // Join lessons with user_progress to find the current 'active' lesson
  $lessonSmt = $pdo->prepare("
      SELECT l.id, l.title, l.category
      FROM lessons l
      LEFT JOIN user_progress up ON l.id = up.lesson_id AND up.user_id = ?
      WHERE up.status IS NULL OR up.status != 'completed'
      ORDER BY l.sort_order ASC
      LIMIT 1
      ");
  $lessonSmt->execute([$user_id]);
  $nextLesson = $lessonSmt->fetch(PDO::FETCH_ASSOC);
  // Fallback if no active lesson (e.g., all completed or none started), find first locked or default

-- Description: This file creates the database, tables, and initial data for the LingoFlow project.
-- Section 0: Database Initialization
-- --------------------------------------------------------------
-- This ensures a clean slate for every run, perfect for development.
DROP DATABASE IF EXISTS `lingoflow_db`;
CREATE DATABASE `lingoflow_db` CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE `lingoflow_db`;
SET
    FOREIGN_KEY_CHECKS = 0;
-- Section 1: Users Table
-- --------------------------------------------------------------
-- Stores core user information, credentials, and top-level stats.
CREATE TABLE `users` (
    `id` INT(11) NOT NULL AUTO_INCREMENT,
    `username` VARCHAR(50) NOT NULL,
    `email` VARCHAR(100) NOT NULL,
    `password` VARCHAR(255) NOT NULL,
    -- Storing plaintext as requested.
    `xp` INT(11) NOT NULL DEFAULT 0,
    `streak` INT(11) NOT NULL DEFAULT 0,
    `accuracy` DECIMAL(5, 2) default 87.00,
    `rank` VARCHAR(50) DEFAULT 'Silver II',
    `role` VARCHAR(20) NOT NULL DEFAULT 'learner',
    `status` VARCHAR(20) NOT NULL DEFAULT 'active',
    `created_at` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    `last_login` TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    PRIMARY KEY (`id`),
    UNIQUE KEY `username` (`username`),
    UNIQUE KEY `email` (`email`)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;
-- Section 2: Lessons Table
-- --------------------------------------------------------------
-- Defines all available lessons in the learning roadmap.
-- The 'id' column uses a string to easily link with front-end JS logic.
CREATE TABLE `lessons` (
    `id` VARCHAR(50) NOT NULL,
    `title` VARCHAR(100) NOT NULL,
    `category` VARCHAR(50) NOT NULL,
    `description` VARCHAR(255) DEFAULT NULL,
    `xp_value` INT(11) NOT NULL DEFAULT 50,
    `sort_order` INT(11) NOT NULL,
    `type` VARCHAR(20) NOT NULL DEFAULT 'lesson',
    PRIMARY KEY (`id`)
) ENGINE = InnoDB DEFAULT CHARSET = utf8mb4;
-- Section 6: Questions Table & Seed Data
-- --------------------------------------------------------------
CREATE TABLE `questions` (
    `id` INT(11) NOT NULL AUTO_INCREMENT,
    `lesson_id` VARCHAR(50) NOT NULL,
    `question_text` VARCHAR(255) NOT NULL,

    // 1. Get raw data from PHP
    const questions = <?php echo json_encode($currentQuestion); ?>;
    const lessonId = "<?php echo $lesson_id; ?>";
    let currentIndex = 0;
    let selectedOption = null;
    let isAnswered = false;
    let lastCorrectPosition = -1; // To track where the correct answer was last time
    /**
     * True Random Shuffle (Fisher-Yates)
     */
    function shuffleArray(array) {
      for (let i = array.length - 1; i > 0; i--) {
        const j = Math.floor(Math.random() * (i + 1));
        [array[i], array[j]] = [array[j], array[i]];
      }
      return array;
    }
    function loadQuestion() {
      isAnswered = false;
      selectedOption = null;
      document.getElementById("feedback").style.display = "none";
      const button = document.getElementById("check-button");
      button.innerText = "Check Answer";
      button.classList.remove("active");
      const q = questions[currentIndex];
      document.getElementById("question-text").innerText = q.q;
      const container = document.getElementById("options-container");
      container.innerHTML = "";
      // --- ABSOLUTE RANDOMIZATION LOGIC ---
      // 1. Copy the answers
      let currentAnswers = [...q.answers];
      // 2. Shuffle them
      currentAnswers = shuffleArray(currentAnswers);
      // 3. (Optional) Pattern Breaker:
      // If the correct answer ended up in the EXACT same visual slot as the previous question,
      // and we have more than 1 option, shuffle again one more time to try and move it.
      // This makes it feel "more random" to humans by avoiding repetition.
      if (currentAnswers.length > 1) {
        let newCorrectPos = currentAnswers.findIndex((a) => a.correct);
        if (newCorrectPos === lastCorrectPosition) {
          // Shuffle one more time to try and dislodge it
          currentAnswers = shuffleArray(currentAnswers);
        }
      }
      // 4. Update tracker for next time
      lastCorrectPosition = currentAnswers.findIndex((a) => a.correct);
      // 5. Render
      currentAnswers.forEach((ans) => {
        const b = document.createElement("button");
        b.className = "option-button";
        b.innerText = ans.text;
        b.onclick = () => {
          if (isAnswered) return;
          document
            .querySelectorAll(".option-button")
            .forEach((el) => el.classList.remove("selected"));
          b.classList.add("selected");
          selectedOption = b;
          button.classList.add("active");
        };
        // Store truth data on the button DOM element
        b.dataset.correct = ans.correct;
        container.appendChild(b);
      });
      // Update progress bar
      document.getElementById("progress-bar").style.width =
        `${(currentIndex / questions.length) * 100}%`;
    }
    function handleCheck() {
      const button = document.getElementById("check-button");
      if (isAnswered) {
        // Go to next question or finish
        currentIndex++;
        if (currentIndex < questions.length) {
          loadQuestion();
        } else {
          // REDIRECT COMMAND
          button.innerText = "Syncing...";
          button.style.opacity = "0.5";
          button.style.pointerEvents = "none";
          window.location.replace(
            `lesson.php?type=${lessonId}&action=complete`,
          );

        return;
      }

      isAnswered = true;
      const isCorrect = selectedOption.dataset.correct === "true";

      selectedOption.classList.add(isCorrect ? "correct" : "wrong");

      feed.style.color = isCorrect ? "var(--green)" : "var(--red)";

      button.innerText = "Continue";
    }
    //Initialize
    loadQuestion();

  header("Location: login.php");

$statusMsg = '';
$statusType = '';

$dbHost = 'localhost';

  if ($_SERVER['REQUEST_METHOD'] === 'POST' && isset($_POST['action'])) {

      $new_email = trim($_POST['new_email'] ?? '');
      if (!empty($new_email) && filter_var($new_email, FILTER_VALIDATE_EMAIL)) {
        // Check if email is already taken by another user
        $stmt = $pdo->prepare("SELECT id FROM users WHERE email = ? AND id != ?");
        $stmt->execute([$new_email, $user_id]);
        if ($stmt->fetch()) {
          $statusMsg = "That email address is already in use.";
          $statusType = "error";
        } else {
          $stmt = $pdo->prepare("UPDATE users SET email = ? WHERE id = ?");
          $stmt->execute([$new_email, $user_id]);
          $statusMsg = "Email address successfully updated.";
          $statusType = "success";
        }
      } else {
        $statusMsg = "Please enter a valid email address.";
        $statusType = "error";
      }

      $new_password = $_POST['new_password'] ?? '';
      $confirm_password = $_POST['confirm_password'] ?? '';
      // Verify current password first
      $stmt = $pdo->prepare("SELECT password FROM users WHERE id = ?");
      $stmt->execute([$user_id]);
      $userData = $stmt->fetch(PDO::FETCH_ASSOC);
      if ($userData['password'] !== $current_password) {
        $statusMsg = "Current password is incorrect.";
        $statusType = "error";
      } elseif ($new_password !== $confirm_password) {
        $statusMsg = "New passwords do not match.";
        $statusType = "error";

        $statusType = "error";
      } else {

        $stmt->execute([$new_password, $user_id]);
        $statusMsg = "Security passcode successfully updated.";
        $statusType = "success";
      }
    }
  }
  // --- 6. FETCH USER DATA (Run after possible updates so UI reflects changes) ---
  $stmt = $pdo->prepare("SELECT email FROM users WHERE id = ?");

    function toggleForm(formId) {
      const form = document.getElementById(formId);
      if (form.classList.contains("active")) {
        form.classList.remove("active");
      } else {
        // Close other open forms before opening a new one
        document
          .querySelectorAll(".hidden-form")
          .forEach((f) => f.classList.remove("active"));
        form.classList.add("active");
        // Focus the first input of the revealed form smoothly
        setTimeout(() => {
          const firstInput = form.querySelector('input:not([type="hidden"])');
          if (firstInput) firstInput.focus();
        }, 100);
      }
    }
